# 03 — 새 스키마

**전면 재수집이 합의됐으므로 마이그레이션 스크립트를 쓰지 않는다.** `db.py`의 `SCHEMA` 상수를 새로 쓰고 DB를 지운 뒤 처음부터 받는다.

> ⚠️ **주석은 반드시 `CREATE TABLE`의 괄호 *안*에 둔다.** SQLite는 문장을 파싱해 다시 쓰면서 괄호 **밖**의 주석을 버리므로, 테이블 위에 적은 설명은 `.schema`에 도달하지 못한다. 이 프로젝트에서 `.schema`가 1차 인터페이스라는 것이 이 규칙의 이유다. `selftest`의 `COMMENT_GUARDS`가 이걸 잠근다.

```sql
CREATE TABLE IF NOT EXISTS committees (
    -- 위원회. **의안·회의록·의원이 전부 이 이름을 들고 있어서 차원 테이블이 필요하다.**
    -- 이게 없으면 bills.committee / meetings.committee_name / bill_stages.committee_name /
    -- member_committees.committee_name 이 각자 문자열을 들고 있고 **넷이 일치한다는 보장이
    -- 아무 데도 없다.** 한쪽에 공백 차이가 하나 생기면 조인이 조용히 0건이 되는데
    -- 잡아 줄 것이 없다. FK가 있으면 그런 값은 애초에 들어오지 못한다.
    --
    -- ⚠️ **FK 때문에 데이터를 버리지 마라.** 수집 중 모르는 위원회명을 만나면 거부하지 말고
    --    여기 먼저 UPSERT 한 뒤 진행한다. FK의 목적은 일관성이지 수집 차단이 아니다.
    --
    -- ⚠️ **위원회의 종류(상임위/특위/예결위)는 여기 없다.** 회의록 트리의 섹션
    --    (상임위원회·예산결산특별위원회·특별위원회·국정감사·국정조사)은 **회의의 성질**이지
    --    위원회의 성질이 아니다 — 국정감사는 상임위별로 열리므로 과방위 회의가 두 섹션에
    --    나타난다. v2 는 그 값을 여기 넣어 **상임위원회 17개 중 14개가 '국정감사'로 굳었다.**
    --    회의의 성질은 meetings.meeting_kind 가 담는다.
    committee_name   TEXT PRIMARY KEY,
                     -- 원천에 적힌 정식 명칭. 약칭으로 바꾸지 마라 — 조인 키다.
                     -- ⚠️ **소위원회는 '상위 위원회 + 공백 + 소위명'이다.**
                     --    '과학기술정보방송통신위원회 정보통신방송법안심사소위원회'
                     --    소위 이름만으로 조회하면 **0건이 온다** — 그리고 0건은 에러가
                     --    아니라 그대로 답이 되어 버린다. 이름을 모르면 추측하지 말고
                     --    SELECT DISTINCT 로 확인하라.
                     -- 왜 수식하나: 소위 이름이 상위를 넘어 중복된다. '법안심사제1소위원회'는
                     --    법사위·행안위·복지위·정무위에 전부 있다. 맨 이름을 키로 쓰면 서로
                     --    다른 위원회가 한 행으로 접혀 "이 소위 회의"가 남의 회의를 섞어 준다.
                     -- ⚠️ **두 원천이 같은 위원회를 다르게 적는다** — likms '기후위기 특별위원회'
                     --    / record '기후위기특별위원회'. upsert_committee 가 공백·중점을 지운
                     --    형태로 기존 행을 찾아 그 표기를 쓰고 **실제로 저장된 이름을 돌려준다.**
                     --    자식 행은 반드시 그 반환값으로 넣어라 — 넘긴 이름 그대로 넣으면 FK 위반이다.
    parent_committee TEXT REFERENCES committees(committee_name)
                     -- 소위원회면 상위 위원회. 아니면 NULL. **이 컬럼이 "소위인가"의 답이다** —
                     -- 별도 플래그를 두지 않는다(v2 의 meetings.is_subcommittee 는 이 값과
                     -- 2,047건 전부 일치했다).
                     -- 회의록 본문 헤더가 '…위원회회의록 (소위원회명)' 형태로 둘을 한 번에 준다.
                     -- 괄호 밖이 여기, 괄호 안이 committee_name 이다.
                     -- ⚠️ **자기참조 FK라 상위 위원회를 먼저 넣어야 한다.** 소위를 먼저 INSERT
                     --    하면 FK 위반이다. 한 회의를 넣는 트랜잭션 안에서 상위 → 소위 순으로 UPSERT 하라.
);

CREATE TABLE IF NOT EXISTS members (
    -- 22대 국회의원. 사퇴·의원직 상실로 떠난 사람도 행을 지우지 않는다 —
    -- 그 사람의 과거 표결·발의·발언이 전부 여기를 참조하고 있어서 지우면 이력이 끊긴다.
    --
    -- **매 실행이 전원을 다시 받는다.** 323명이라 몇 분이면 끝나므로 "언제 확인했나"를
    -- 저장할 이유가 없다. 그래서 타임스탬프 컬럼이 없다.
    open_na_id      TEXT PRIMARY KEY,
                    -- 'KIMHyun'. 의원 상세 URL(https://www.assembly.go.kr/members/22/{이값})의
                    -- 마지막 조각이고, **표결 명단·발의자 명단·회의록 발언에 나오는 유일한 식별자다.**
                    -- ⚠️ 대소문자 규칙이 없다: KIMHyun · CHOIMinhee · KANGKYUNGSOOK · LEEHAIMIN.
                    --    이름에서 만들어낼 수 없으므로 반드시 원천의 href 에서 가져온다.
                    -- ⚠️ **두 호스트가 같은 사람을 다르게 로마자화한다.** record 의 회의록은
                    --    JUNGHYEKYUNG 으로 링크하는데 www 의 실제 slug 는 JEONGHYEKYEONG 이고
                    --    앞의 것은 404다(실측 3명). merge_dead_member_slugs 가 **죽은 slug 만**
                    --    합친다 — 이름으로 합치면 동명이인(22대에 박지원 2명)이 한 사람이 된다.
    name            TEXT NOT NULL,           -- '김현'
                    -- ⚠️ 조인 키로 절대 쓰지 마라. 동명이인이 실재한다 —
                    --    원천 사이트에 동명이인 전용 팝업(selSameNmMemb.do)이 따로 있는 것이 증거다.
    party           TEXT,                    -- 최신 정당. 덮어쓴다 — 변경 이력은 두지 않는다.
                    -- 이력을 안 두는 이유: 유일한 실제 쓰임은 "그 표결 당시 어느 당이었나"인데
                    -- 수집 시작 **이전**의 변경은 웹사이트에 없어서 채울 수 없다. 즉 그 질문에
                    -- 답하지 못하는 채로 테이블만 늘어난다.
                    -- ⚠️ NULL 이면 조용히 샌다. `GROUP BY party` 가 그 의원을 NULL 버킷으로
                    --    몰아넣어 정당별 집계에서 빠지는데 에러가 안 난다. 감사의
                    --    members_missing_party 가 그 수를 센다.
    district        TEXT,                    -- '경기 안산시을'. **비례대표는 NULL이다.**
                    -- 지역구/비례를 가르는 별도 컬럼을 두지 않는다 — 이 컬럼의 NULL 여부가 그 답이다.
                    -- 정규화하지 마라. '전남광주통합특별시 순천시광양시곡성군구례군갑'처럼
                    -- 행정구역 통합이 반영된 긴 이름이 정상이다.
    is_incumbent    INTEGER NOT NULL DEFAULT 1
                    -- 판정은 **의원 상세 페이지의 `jeonzik_name` 클래스 유무**로 한다.
                    -- 전직이면 그 클래스가 붙고 현직이면 없다(실측: 전재수 4회 / 김현 0회).
                    -- 손으로 관리하지 않고 매 갱신이 다시 판정한다.
                    -- ⚠️ 0인 행을 지우지 마라. 22대 중 떠난 21명의 과거 표결·발의·발언이
                    --    전부 이 행을 참조한다. 지우면 그 기록이 조인에서 통째로 사라진다.
                    --    22대를 거쳐 간 총원은 320명 안팎(현직 299 + 전직 21)이다.
);

CREATE INDEX IF NOT EXISTS idx_members_name  ON members(name);
CREATE INDEX IF NOT EXISTS idx_members_party ON members(party);

CREATE TABLE IF NOT EXISTS member_committees (
    -- 의원의 소속 위원회. 한 의원이 둘 이상일 수 있어(예: 과방위 + 예결위) 별도 테이블이다.
    -- "이 법안을 심사하는 위원회의 의원들은 누구인가"와, 회의록 발언자를 읽을 때
    -- "이 사람은 이 위원회 소속이다"에 답한다.
    open_na_id      TEXT NOT NULL REFERENCES members(open_na_id) ON DELETE CASCADE,
    committee_name  TEXT NOT NULL REFERENCES committees(committee_name),
    PRIMARY KEY (open_na_id, committee_name)
);

CREATE TABLE IF NOT EXISTS bills (
    -- 22대에 제출된 **모든 의안**. 법률안뿐 아니라 예산안·결의안·동의안·규칙안까지 들어온다.
    --
    -- 왜 전체를 담나: 의안번호가 종류 구분 없이 연속된 정수라서, 전체를 담아야
    -- "번호에 구멍이 없다 = 누락이 없다"가 성립한다. 법률안만 담으면 구멍이 생기고
    -- 그 구멍이 비법률안인지 수집 실패인지 영원히 구분할 수 없다.
    bill_no         TEXT PRIMARY KEY,
                    -- '2214631'. '22' + 5자리 일련번호. 사람이 읽을 수 있고
                    -- 보도자료와 대조되는 유일한 키라 PK로 삼았다.
                    -- ⚠️ INTEGER 가 아니라 TEXT 인 이유: 'ZZ22124' 형식이 88건 있다
                    --    (국정조사계획서·기금운용평가보고서 등 의안번호 체계 밖의 문서).
                    --    7자리 고정폭이라 TEXT 여도 사전순 = 번호순이다.
                    --    완결성 검증은 숫자인 것만 대상으로 한다.
    bill_id         TEXT NOT NULL UNIQUE,
                    -- 'PRC_A2I6…' 또는 'ARC_Z2Z6…'. 상세·표결·발의자 요청에 넣는 핸들.
                    -- 상세 URL: https://likms.assembly.go.kr/bill/bi/billDetailPage.do?billId={이값}
                    -- ⚠️ 접두어가 둘이다. 전수 실측 PRC_ 20,128건 · ARC_ 470건(정부제출 쪽).
                    --    'PRC_'만 가정하는 정규식을 쓰면 470건이 조용히 사라진다.
                    -- ⚠️ 영구키로 가정하지 마라. 동일성 판단의 기준은 언제나 bill_no 다.
    bill_kind       TEXT NOT NULL,           -- '법률안' | '예산안' | '결의안' | '동의안' | …
                    -- 상세의 billKindCd 에서 온다. **목록에는 이 값이 없다** —
                    -- 목록 표의 컬럼은 여덟 개고 의안종류가 거기 없다.
                    -- ⚠️ 목록 UPSERT 가 이 컬럼을 덮으면 안 된다. v1 이 그렇게 20,598건 전부를
                    --    '미상'으로 되돌렸고, 그 사이 bill_detail_missing 게이트가
                    --    **아무것도 검사하지 않으면서 초록불**이었다.
    title           TEXT NOT NULL,           -- 의안명
                    -- ⚠️ 목록 셀 앞에 '계류의안'/'처리의안' 배지 텍스트가 붙어 있다. 떼고 저장한다.
                    -- 위원장 발의는 끝 괄호에 위원회가 들어간다: '…법률안(대안)(농림축산식품해양수산위원장)'.
                    -- 그 값은 언제나 committee 와 같으므로(실측 933/934, 불일치 0) 따로 담지 않는다.
    proposer_kind   TEXT,                    -- '의원' | '위원장' | '정부' | '의장' | '기타'
                    -- ⚠️ **'의원'이 아니면 bill_proposers 에 행이 없다.** 위원장 934건 ·
                    --    정부 601건 · 의장 56건 · 기타 63건. 그리고 하필 법이 되는 것의
                    --    대부분이 위원장 대안이다 — "정당별 통과 법안 수"를 물으면
                    --    반드시 이 컬럼으로 그들을 따로 세라.
                    -- 위원장 발의의 주체는 committee 가 답한다. 정부는 원천이 부처를 안 준다
                    -- (상세의 제안자 칸이 '정부' 세 글자다). 그래서 이 컬럼이 정부 발의에 대해
                    -- 알 수 있는 전부다.
    date_proposed   TEXT,                    -- 제안일자. 위젯 '접수' 단계의 날짜와 같다.
    committee       TEXT REFERENCES committees(committee_name),
                    -- **지금** 이 의안이 걸려 있는 소관위원회. JSON `currCmt` 가 직접 준다.
                    -- 거쳐 온 이력은 bill_stages 가 갖는다 — 위원회 이관이 464건 있다
                    -- (행안위 → 정치개혁특위 → 행안위 같은 왕복).
                    -- ⚠️ HTML 표에서 유도하지 마라. 원천이 JSON 으로 직접 말해 주는 값이고,
                    --    DOM 이 바뀌면 깨지는 쪽을 굳이 고를 이유가 없다.
    reason_text     TEXT,                    -- 제안이유 및 주요내용 (평문)
                    -- 키워드 검색의 주 대상. 조문 원문(HWP/PDF)은 수집하지 않는다.
                    -- ⚠️ 이 값이 NULL 인 것을 '상세 미수집'의 판정자로 쓰지 마라.
                    --    제안이유가 원래 비어 있는 의안이 있고(결의안 220건 중 216건),
                    --    그러면 그 의안은 영원히 미수집으로 잡혀 매 실행이 자기를 다시 받는다.

    -- ── 진행과 결말: 원천의 '심사진행단계' 위젯을 그대로 옮긴 둘 ──────────
    -- 이 둘이 v3 의 중심이다. v2 는 같은 축을 네 컬럼이 서로 다른 어휘로 말했고
    -- (review_status 18값 · decision_result 7값 · outcome 9값 · bill_stages.result 15값),
    -- 그 사이를 잇는 derive_outcome() 이 모르는 값을 조용히 '계류'로 떨어뜨려
    -- **공포된 법안 1건을 계류로 판정하고 있었다**(2200851, 재의 가결 후 공포).
    -- v3 은 원천이 화면에 그리는 값을 그대로 저장한다 — 유도 규칙이 없으면 유도 버그도 없다.
    current_stage   TEXT,
                    -- **심사진행단계 위젯에서 날짜가 적힌 맨 오른쪽 칸의 라벨.**
                    -- 값: 접수 · 위원회 심사 · 체계자구 심사 · 본회의 심의 · 정부이송 ·
                    --     정부재의 · 공포 · 대안반영폐기 · 수정안반영폐기 · 철회 · 폐기 · 부결
                    --
                    -- 원천: POST /bill/bi/bill/detail/billInfo.do → #insc-step-sect > div[data-gbn]
                    --   RECEIVE 접수 / CMT 위원회 심사 / EXAM 체계자구 심사 / MAIN 본회의 심의 /
                    --   GOVTRANS 정부이송 / REEXAM 정부재의 / DCS (결말·라벨 가변)
                    --   class: "" 완료 · "on" 현재 · "proc" 예정(날짜 없음)
                    --
                    -- ⚠️ **CHECK 를 걸지 않는다.** 국회는 새 라벨을 예고 없이 늘리고, CHECK 를
                    --    걸면 그때 수집이 통째로 멈춘다. 대신 감사가 DISTINCT current_stage 를
                    --    보고값으로 내므로 못 보던 값이 눈에 띈다.
                    -- ⚠️ **단계 목록의 길이는 의안마다 다르다.** 부결 의안에 '위원회 심사'가
                    --    없기도 하고(2217099), 대안반영폐기는 세 칸에서 끝난다. "여섯 칸이 다
                    --    있다"고 가정하는 파서를 쓰지 마라.
    outcome         TEXT NOT NULL DEFAULT '계류'
                    CHECK (outcome IN ('계류','공포','대안반영폐기','수정안반영폐기',
                                       '철회','폐기','부결')),
                    -- "이 법안은 결국 어떻게 됐나"의 **단일 답**. 위젯의 DCS 라벨이다.
                    --
                    -- 판정 규칙 (구현은 04번 문서):
                    --   ① DCS 칸에 날짜가 있다        → 그 라벨
                    --   ② 날짜가 없는데 alt_bill_no    → '대안반영폐기'
                    --   ③ 그 밖                        → '계류'
                    --
                    -- ⚠️ **'본회의불부의'는 값에 없다 — '대안반영폐기'에 병합했다.** 원천 목록은
                    --    둘을 구분해 적지만 실측하면 같은 운명의 두 시점이다: 본회의불부의
                    --    319건 중 317건이 대안 관계를 갖고 **그 대안의 결말이 예외 없이 '계류'**
                    --    (= 아직 본회의 전)인 반면, 대안반영폐기 3,839건의 대안은 공포 3,692 ·
                    --    재의부결 122 · 계류 25 · 부결 2다. 위젯이 본회의불부의를 안 보여 주는
                    --    것도 같은 이유다 — 그건 결말이 아니라 진행 중인 상태다.
                    --    ②번 규칙이 이 부류를 정확히 집는다.
                    -- ⚠️ **'재의부결'도 값에 없다.** 원천 위젯은 그냥 DCS:부결 이고 앞에
                    --    REEXAM:정부재의 가 붙을 뿐이다(실측 2208496·2208485·2200038).
                    --    거부권 뒤 부결된 법안은 `outcome='부결' AND date_reexam_requested
                    --    IS NOT NULL` 로 찾는다. 반대로 거부권 뒤 재의결로 법이 된 것은
                    --    `outcome='공포' AND date_reexam_requested IS NOT NULL` 이다.
                    -- ⚠️ **'임기만료폐기'도 없다.** 22대가 끝나 그 상태가 실제로 생기면
                    --    그때 넣어라 — 규칙 없는 값은 조회하는 쪽에 거짓 기대만 만든다.
    alt_bill_no     TEXT,
                    -- 이 의안을 흡수한 **대안의 의안번호**. 대안 관계가 없으면 NULL.
                    -- 역방향은 이 컬럼의 역조회다: "이 대안이 무엇들을 흡수했나" =
                    --   SELECT bill_no FROM bills WHERE alt_bill_no = ?
                    --
                    -- ⚠️ **FK 를 걸지 않는다** — 대안이 원안보다 늦게 접수되는 것이 정상이라
                    --    전방 참조다. 대신 감사가 "alt_bill_no 가 bills 에 없는 행 수 = 0"을 센다.
                    -- ⚠️ **이 값은 자기 페이지에서 안 올 수 있다.** 본회의불부의 상태의 의안은
                    --    대안정보 탭이 아예 없다(실측 TABS: []). 그 관계는 **대안 쪽**의
                    --    anBillInfo.do 가 흡수한 원안들을 열거하면서 채워진다. 대안은 원안보다
                    --    번호가 크므로 같은 수집 패스 안에서 반드시 뒤에 온다.
                    -- v2 는 이걸 bill_alternatives 테이블로 뒀는데, 형제 목록을 양방향으로 쓰는
                    --    로직이 **무관한 대안을 2건 끼워 넣었다**(2200057 → 위안부 피해자법 대안).
                    --    컬럼은 값을 두 개 못 담아 그 오염이 구조적으로 불가능하다.

    -- ── 마일스톤: 위젯 단계 중 위원회가 없는 것들 ──────────
    date_transferred      TEXT,              -- 위젯 '정부이송'. JSON `govTransDt`.
    date_reexam_requested TEXT,              -- 위젯 '정부재의'. 대통령 거부권 행사일.
                    -- ⚠️ **원천에서 이 값이 오는 곳은 위젯뿐이다.** 상세의 어떤 표에도
                    --    재의 칸이 없고 pre.bill_memo 의 산문에만 남는다. v2 는 그 산문을
                    --    정규식으로 긁어 reconsideration_result 를 만들었는데, 위젯이 날짜를
                    --    직접 주므로 그럴 필요가 없어졌다. 22대 실측 28건.
    date_promulgated      TEXT,              -- 위젯 '공포'. JSON `announceDt`.
                    -- 공포된 법률의 **이름**은 bill_promulgated_laws 에 있다 — 일괄개정은
                    -- 의안 하나가 법률 여러 개를 공포하므로 컬럼 하나로는 담기지 않는다.

    detail_collected_at   TEXT
                    -- 상세를 받은 시각. 상세 트랜잭션이 커밋될 때만 값이 들어가므로
                    -- 값이 있다는 것은 그 전부(단계·발의자·표결·대안·공포법률)가 들어왔다는 뜻이다.
                    -- ⚠️ **재수집 대상 판정에는 쓰지 않는다.** 그건 `outcome='계류'` 한 줄이다
                    --    (04번 문서). 이 컬럼의 쓰임은 둘이다 — 수집이 중간에 죽었을 때 어디까지
                    --    왔는지 보는 것과, 감사가 "상세를 한 번도 안 받은 의안"을 세는 것.
);

CREATE INDEX IF NOT EXISTS idx_bills_stage    ON bills(current_stage);

CREATE INDEX IF NOT EXISTS idx_bills_alt      ON bills(alt_bill_no);
-- idx_bills_title 은 두지 않는다. 제목 검색은 언제나 `LIKE '%…%'` 라 접두가 없고,
-- 접두 없는 LIKE 는 인덱스를 못 쓴다. 매 INSERT 마다 쓰기 비용만 내던 인덱스였다.

CREATE TABLE IF NOT EXISTS bill_stages (
    -- 심사 단계. **상태 컬럼이 아니라 사건 행으로 쌓는다.**
    --
    -- 왜 행인가: 같은 단계가 여러 번 있다. 소위가 둘인 법안이 있고, **위원회 이관이 464건**
    -- 있다 — 2200027 은 행안위(2024-06-11 회부 → 2026-03-12 처리) → 정치개혁특위
    -- (2026-03-12 → 2026-06-01) → 행안위(2026-07-01 회부, 심사 중)로 돌아왔다.
    -- bills 의 컬럼 몇 개로는 접히지 않는다.
    --
    -- ⚠️ **bills.current_stage 와 다른 것을 답한다.** current_stage 는 원천 위젯이 요약한
    --    "지금 어디"이고 이 테이블은 "어느 위원회에 언제 들어가 언제 나왔나"의 원장이다.
    --    위젯의 CMT 한 칸이 여기서는 여러 행일 수 있다.
    --
    -- ⚠️ 조회할 때 **언제나 LEFT JOIN 이다.** 대안·정부제출 법안은 소관위 회부 단계 자체가
    --    없어 행이 아예 없거나 적다. INNER JOIN 하면 하필 법이 될 가능성이 가장 높은
    --    법안들이 빠진다.
    bill_no         TEXT NOT NULL REFERENCES bills(bill_no) ON DELETE CASCADE,
    stage           TEXT NOT NULL
                    CHECK (stage IN ('소관위','소위','체계자구','본회의')),
                    -- 원천의 '심사정보' 표 하나가 여기 한 행이다.
                    --   소관위 심사정보 → '소관위'   /  소위 심사정보 → '소위'
                    --   법사위 체계자구 심사정보 → '체계자구'  /  본회의 심의정보 → '본회의'
                    -- '접수'·'정부이송'·'공포'·'대안반영폐기'는 여기 없다 — 위원회도 회부일도
                    --    없는 한 번짜리 사건이라 bills 의 컬럼과 current_stage 가 담는다.
                    -- '관련위'도 없다 — 원천이 위원회명과 회부일만 주고 상정일·의견서제시일·
                    --    처리결과를 **칸은 두고 값을 안 보낸다.** 담아도 자라지 않는 사실이다.
    seq             INTEGER NOT NULL DEFAULT 1,
                    -- 같은 단계의 몇 번째인가. 위원회 이관·소위 복수 회부를 가른다.
                    -- ⚠️ **이 값의 안정성에 멱등성을 걸지 마라.** 소위는 절반(53%)이
                    --    committee_name 이 비어 정렬 타이가 생기고, 타이가 뒤집히면 같은
                    --    사건이 다른 seq 를 받는다. 그러면 PK 가 안 부딪혀 UPSERT 가
                    --    덮어쓰기가 아니라 **삽입**으로 동작하고 에러 없이 행만 는다.
                    -- 그래서 이 테이블은 UPSERT 를 쓰지 않는다. 의안 하나의 단계 전부를
                    --    **한 트랜잭션 안에서 DELETE 후 INSERT** 한다(replace_children_txn).
    committee_name  TEXT REFERENCES committees(committee_name),
                    -- ⚠️ 본회의 단계는 원천에 위원회 칸이 **아예 없다**. NULL 이 정상이다.
                    -- ⚠️ 소위는 원천이 이름을 안 주는 경우가 절반이다(실측 7,790/16,434 = 47%).
                    --    **이것만이 진짜 결측이다** — 아래 표의 다른 NULL 과 뜻이 다르다.
                    -- ⚠️ 체계자구는 표에 위원회 칸이 없고 caption('법사위 체계자구 심사정보')이
                    --    곧 그 값이다. v1 은 이 칸을 안 채워 1,501행 전부 NULL 이었다.
    date_referred   TEXT,                    -- 회부일
                    -- **이 컬럼이 이 테이블이 존재하는 이유의 정중앙이다.** 계류 법안
                    -- 14,788건 중 처리일이 하나라도 있는 것은 994건(6.7%)뿐이고, 나머지
                    -- 13,794건은 회부일이 유일한 시간 표지다. 그중 10,990건이 소위에 회부돼
                    -- 있다 — 회부일이 없으면 "소위까지 간 법안"과 "접수만 된 법안"이 구별되지 않는다.
    date_processed  TEXT,                    -- 처리일 (본회의는 의결일)
    PRIMARY KEY (bill_no, stage, seq)
    -- ⚠️ **같은 NULL 이 단계마다 다른 뜻이다. 이 표 없이 SQL 을 쓰면 에러 없이 틀린다.**
    --    (2026-08-10 실측)
    --
    --      단계      행수     위원회        회부일   처리일
    --      ──────────────────────────────────────────────
    --      소관위    20,703   100%          95%      32%
    --      소위      16,434   47%           98%      32%
    --      체계자구   1,501   100%(법사위)  100%     97%
    --      본회의     5,885   0%(칸 없음)   0%       100%
    --
    --    **본회의의 위원회·회부일은 원천에 칸이 아예 없다** — 0%는 우리가 못 읽은 것이
    --    아니라 그런 사실이 없는 것이다. 반대로 낮은 값(소관위 처리 32%)은 *아직 그
    --    사건이 안 일어난 것*이고, 그 사이가 계류 14,788건의 실체다.
    --    **소위의 위원회 47%만이 진짜 결측이다.**
    --
    -- ⚠️ **처리결과(원안가결·수정가결·대안반영폐기·회송…)는 담지 않는다.** outcome 을
    --    원천 위젯이 직접 주므로 유도에 쓰이지 않고, 안 담으면 "위원회 어휘와 본회의
    --    어휘가 다르다"는 함정도 같이 사라진다. 위원회가 되돌려보낸 것(회송)은 다음 행의
    --    회부일로 드러난다.
    -- ⚠️ **상정일(presentDt)도 담지 않는다.** 원천에는 있지만 이 DB 의 목적(어디까지 왔나 +
    --    그 과정의 회의 발언)에 비추어 지엽적이라고 판단했다.
);

CREATE INDEX IF NOT EXISTS idx_stages_cmt ON bill_stages(committee_name, date_referred);

CREATE TABLE IF NOT EXISTS bill_promulgated_laws (
    -- 이 의안으로 공포된 법률들. **의안당 여러 행이다.**
    --
    -- 왜 bills 의 컬럼이 아닌가: 일괄개정 의안 하나가 법률 여러 개를 한꺼번에 공포한다
    -- (실측 11건이 2~10개, 최대 2201193 의 10개). 컬럼 하나로 두면 나머지가 조용히 버려지고,
    -- 버려진 줄 아무도 모른다. 공포일은 의안 단위 사실이라 bills.date_promulgated 에 있다.
    bill_no         TEXT NOT NULL REFERENCES bills(bill_no) ON DELETE CASCADE,
    law_name        TEXT NOT NULL,
                    -- 공포법률명.
                    -- ⚠️ **의안명과 다르다.** 의안명 '…기본법안(대안)' → 공포법률명 '…기본법'.
                    --    '안(대안)' 꼬리가 떨어지고, 이름 자체가 바뀌기도 한다
                    --    (저출산·고령사회기본법 → **인구전략기본법**). 법제처 등 바깥 자료와
                    --    이름으로 맞출 때는 반드시 이쪽을 쓴다.
    law_no          TEXT,                    -- 공포번호
    PRIMARY KEY (bill_no, law_name)
                    -- ⚠️ seq 를 키로 쓰지 않는 이유는 bill_stages.seq 의 그것과 같다 —
                    --    순서로 키를 만들면 원천이 행을 재정렬할 때 같은 사실이 다른 키를 받고,
                    --    그러면 재수집이 덮어쓰기가 아니라 삽입이 된다. 한 의안이 같은 법률을
                    --    두 번 공포하지 않으므로 이름이 안정된 키다.
);

CREATE INDEX IF NOT EXISTS idx_laws_name ON bill_promulgated_laws(law_name);

CREATE TABLE IF NOT EXISTS bill_proposers (
    -- 발의자. **의원만 담는다.**
    --
    -- ⚠️ **이 테이블만 조인하면 가결 법안의 64%가 조용히 사라진다.** 위원장 대안(934건)·
    --    정부제출(601건)·의장(56)·기타(63)에는 개별 의원 발의자가 없다. 그리고 하필 법이
    --    되는 것은 대부분 위원장 대안 쪽이다. "정당별 통과 법안 수"를 물으면 반드시
    --    bills.proposer_kind 로 그들을 따로 세라 — 위원장 발의의 주체는 bills.committee 다.
    bill_no         TEXT NOT NULL REFERENCES bills(bill_no) ON DELETE CASCADE,
    open_na_id      TEXT NOT NULL REFERENCES members(open_na_id),
    role            TEXT NOT NULL CHECK (role IN ('대표발의','공동발의')),
                    -- ⚠️ **대표발의가 한 명이라고 가정하지 마라 — 공동대표발의가 있다.**
                    --    원천이 팝업의 숨은 필드로 직접 말해 준다:
                    --    `<input id="info" value="최형두의원ㆍ이준석의원ㆍ황정아의원 등 11인">`.
                    --    v1 은 상세 HTML 의 **첫 번째** /members/ 링크만 대표로 잡아서
                    --    role='대표발의' 가 2명 이상인 의안이 **0건**이었다 — 있는데 없는 것으로
                    --    보였고, 0건은 에러가 아니라 그대로 답이 됐다.
    PRIMARY KEY (bill_no, open_na_id)
                    -- 한 의안에서 한 사람이 대표이면서 공동일 수는 없으므로 role 은 키가 아니다.
);

CREATE INDEX IF NOT EXISTS idx_proposers_member ON bill_proposers(open_na_id, role);

CREATE TABLE IF NOT EXISTS bill_vote_summary (
    -- 본회의 표결 요약. 의안당 최대 1행.
    -- 표결 결과(원안가결/수정가결/부결)는 여기 없다 — bills.outcome 과 bill_stages(본회의)가
    -- 이미 담고 있고, v2 에서 셋이 1,656건 전부 일치해 순수 중복이었다.
    bill_no         TEXT PRIMARY KEY REFERENCES bills(bill_no) ON DELETE CASCADE,
    date_voted      TEXT,
    total_seats     INTEGER,                 -- 재적
    present         INTEGER,                 -- 재석
    yes             INTEGER,                 -- 찬성
    no              INTEGER,                 -- 반대
    abstain         INTEGER                  -- 기권
                    -- ⚠️ **불참 수는 재적 − 재석으로 계산한다.** 명단으로 유도하지 마라 —
                    --    아래 bill_votes 참조.
);

CREATE TABLE IF NOT EXISTS bill_votes (
    -- 의원별 표결. **불참은 여기 없다.**
    --
    -- ⚠️ 행이 없다 ≠ 불참. 원천이 찬성·반대·기권 명단만 주고 불참자 명단은 주지 않는다.
    --    재적 299 / 재석 196이면 103명이 안 나온 것인데 그 103명이 누구인지는 알 수 없다.
    -- ⚠️ **명단이 요약보다 1명 적은 의안이 61건 있고, 이건 고칠 수 없다.** 원천이 표결
    --    명단에 이름만 주는데 22대에 박지원이 둘이라(PARKJIEWON 5선 전남 / PARKJIWON
    --    초선 전북) 가를 수 없다. collect_failures 에 gone 으로 남고 감사는 보고값으로 낸다.
    bill_no         TEXT NOT NULL REFERENCES bills(bill_no) ON DELETE CASCADE,
    open_na_id      TEXT NOT NULL REFERENCES members(open_na_id),
    vote            TEXT NOT NULL CHECK (vote IN ('찬성','반대','기권')),
    PRIMARY KEY (bill_no, open_na_id)
);

CREATE INDEX IF NOT EXISTS idx_votes_member ON bill_votes(open_na_id, vote);

CREATE TABLE IF NOT EXISTS bill_meetings (
    -- **의안 → 회의 다리.** 이 테이블의 존재 이유는 하나다 —
    -- "이 법안이 심사되는 과정의 회의에서 무슨 말이 오갔나"에 답하는 것.
    --
    --   bills → bill_meetings → meetings → meeting_utterances
    --
    -- 회의명·회의일·회의결과를 담지 않는다. meetings 가 회기·차수·위원회·날짜를 구조화해
    -- 갖고 있으므로 조인하면 나온다.
    bill_no         TEXT NOT NULL REFERENCES bills(bill_no) ON DELETE CASCADE,
    conference_id   INTEGER NOT NULL,        -- record.assembly.go.kr 의 회의록 id
                    -- 회의록 URL: https://record.assembly.go.kr/assembly/viewer/minutes/xml.do?id={이값}&type=view
                    -- ⚠️ **meetings 에 FK 를 걸지 않는다.** 본회의 회의록은 수집 범위 밖이라
                    --    meetings 에 행이 없고(실측 5,651행), FK 를 걸면 그 연결 정보까지 잃는다.
                    --    본회의 행은 "이 법안을 의결한 본회의 회의록의 id"라는 포인터로만 남는다 —
                    --    본문은 없지만 위 URL 로 사람이 열 수 있다.
    stage           TEXT,                    -- '소관위' | '체계자구' | '본회의'
                    -- 어느 단계의 회의인가. 소위 회의는 원천이 소관위 회의정보 표에 함께 넣으므로
                    -- '소관위'로 들어온다 — 그 회의가 소위였는지는 meetings.committee_name 의
                    -- 상위 위원회 여부(committees.parent_committee)가 답한다.
    PRIMARY KEY (bill_no, conference_id)
);

CREATE INDEX IF NOT EXISTS idx_bill_meetings_conf ON bill_meetings(conference_id);

-- ═══════════════════════════════════════════════════════════
-- 회의록
-- ═══════════════════════════════════════════════════════════

CREATE TABLE IF NOT EXISTS meetings (
    -- 우리가 존재를 아는 모든 회의. 본문 수집 여부와 무관하게 행이 생긴다.
    -- 본문을 받았는지는 meeting_utterances 에 행이 있는지로 판정한다 (상태 컬럼 없음).
    --
    -- **본회의는 수집하지 않는다.** 회의록 트리에 국회본회의 섹션이 있지만 범위 밖으로 뒀다 —
    -- 본회의는 표결하는 자리이고 실질 논의는 상임위·소위에서 일어난다. 표결 숫자와
    -- 의원별 표는 의안 쪽(bill_vote_summary·bill_votes)에서 이미 온다.
    conference_id   INTEGER PRIMARY KEY,     -- xml.do?id= / pdf.do?id= 의 그 id
    meeting_kind    TEXT NOT NULL
                    CHECK (meeting_kind IN
                      ('상임위원회','예산결산특별위원회','특별위원회','국정감사','국정조사')),
                    -- **이 회의가 무엇인가.** 위원회가 무엇인가가 아니다 —
                    -- 국정감사는 상임위별로 열리므로 과방위 회의가 '상임위원회'로도
                    -- '국정감사'로도 나타난다. v2 는 이 값을 committees 에도 써서
                    -- **상임위원회 17개 중 14개가 '국정감사'로 굳었다.**
                    -- ⚠️ **유일한 출처가 회의록 트리의 class_id_sch(2,3,4,5,6)다.**
                    --    본문 헤더 어디에도 이걸 가르는 문자열이 없다(실측). 그래서 열거
                    --    단계에서 id 와 섹션을 쌍으로 들고 와야 한다 —
                    --    "목록은 id 만 열거한다"는 원칙의 유일한 예외다.
                    -- '본회의'는 값에 없다. 수집 범위 밖이다.
                    -- '기타'도 없다. 여섯 번째 값을 열어 두면 파서가 분류에 실패한 회의를
                    --    조용히 거기 버리게 된다.
    session_no      INTEGER,                 -- 432 (제432회)
    sitting_no      INTEGER,                 -- 3 (제3차)
                    -- ⚠️ 차수는 제목 줄이 아니라 <p class="num"> 에 있고 단위가 '차'가 아니라
                    --    '호'다('제3호').
                    -- ⚠️ **국정감사(class 5)만 회기도 차수도 없다** — '2024년도 국정감사
                    --    …위원회회의록'. 그 부류는 둘 다 NULL 인 것이 정상이다(실측 317건).
                    --    회의 누락을 차수 연속성으로 검사할 때 이 부류는 검사 밖이다.
    committee_name  TEXT NOT NULL REFERENCES committees(committee_name),
                    -- 소위원회 회의면 소위원회 이름이 들어온다.
                    -- 상위 위원회는 committees.parent_committee 로 따라간다 —
                    -- **그 값이 NULL 이 아니면 이 회의가 소위 회의다.** 별도 플래그를 두지 않는다.
                    -- 판정: 본문 헤더 제목 줄에서 **'회의록' 뒤에 오는 괄호**.
                    --   '국회운영위원회회의록 (국회운영개선소위원회)' → 괄호 안이 committee_name,
                    --   괄호 밖이 parent_committee.
                    -- ⚠️ **"괄호가 있으면 소위"로 만들면 틀린다.** 위원회명 자체에 괄호가 들어간다:
                    --    '대법관(노경필·박영재·이숙연)임명동의에관한인사청문특별위원회회의록' (id 52240).
                    --    소위 괄호는 언제나 '회의록' 뒤이므로 거기에 앵커를 잡아라. 잘못 잡으면
                    --    없는 위원회가 committees 에 자동 등록되고, FK 는 UPSERT 로 등록된 값을
                    --    막아 주지 않으므로 아무도 알아채지 못한다.
    date_meeting    TEXT NOT NULL
                    -- ⚠️ meeting_kind 를 뺀 나머지 메타는 **회의록 본문 헤더**에서 파싱한다.
                    --    트리의 위원회명·회기·차수에 기대면 트리 구조가 바뀔 때 함께 깨진다.
);

CREATE INDEX IF NOT EXISTS idx_meetings_date ON meetings(date_meeting);
CREATE INDEX IF NOT EXISTS idx_meetings_cmt  ON meetings(committee_name, date_meeting);

CREATE TABLE IF NOT EXISTS meeting_utterances (
    -- 발언. 회의 안에서의 순서를 보존한다. **화자 정보를 여기 함께 담는다.**
    --
    -- v2 는 meeting_speakers 를 따로 두고 회의 안 자리번호(speaker_no)로 이었는데,
    -- 그 번호가 "사람의 식별자가 아니고 회의 밖에서 참조하면 안 되는 값"이라 읽는 쪽이
    -- 매번 배워야 하는 개념이었다. 합치면 그 개념이 통째로 사라진다.
    -- 비용은 +38MB(633MB 기준 +6%) — 테이블 하나와 두 홉 조인을 없애는 값으로 싸다.
    conference_id    INTEGER NOT NULL REFERENCES meetings(conference_id) ON DELETE CASCADE,
    seq              INTEGER NOT NULL,       -- 회의 안에서의 발언 순서 (1부터)
    speaker_name     TEXT NOT NULL,          -- 회의록에 적힌 이름 그대로
    speaker_position TEXT,                   -- '소위원장' · '과학기술정보통신부제2차관' · '수석전문위원'
                     -- 직위는 사람의 속성이 아니라 (회의 × 사람)의 속성이다 —
                     -- 김현 의원은 그 소위에서 '소위원장'이지만 과방위 전체회의에서는 '위원'이다.
    open_na_id       TEXT REFERENCES members(open_na_id),
                     -- ⚠️ **NULL 이면 국회의원이 아니다** (장관·차관·청장·전문위원·증인·참고인).
                     --    실측 화자 40,137명 중 14,898명(37%)이 비의원이다.
                     --    판별은 회의록 HTML 에서 .man 안에 /members/ 링크가 있는지로 한다 —
                     --    **직위 문자열로 분류하지 마라.** 종류가 열려 있고 회의마다 새로 생긴다.
    text             TEXT NOT NULL,          -- 발언 본문. 원천의 여러 문단을 개행으로 이어 붙인다.
    PRIMARY KEY (conference_id, seq)
                     -- ⚠️ **bill_no 컬럼을 여기 두지 마라.** 회의에 상정된 안건은 알 수 있지만
                     --    **개별 발언이 어느 안건에 대한 것인지는 알 수 없다.** 의원마다 질의에서
                     --    다루는 순서가 다르고, 안건을 하나씩 끝내고 넘어가는 방식이 아니다.
                     --    그 컬럼은 채우는 순간 거짓이 된다.
);

CREATE INDEX IF NOT EXISTS idx_utterances_member ON meeting_utterances(open_na_id);

CREATE TABLE IF NOT EXISTS meeting_agenda (
    -- 그 회의에 상정된 안건 목록. **회의 단위 사실이다.**
    -- bill_meetings 와 방향이 다르다 — 저쪽은 "이 법안이 어느 회의에서 다뤄졌나",
    -- 이쪽은 "이 회의에서 무엇을 다뤘나". 후자는 발언을 읽을 때의 맥락이 되고,
    -- **의안이 아닌 안건 6,097건은 여기에만 있다**(청원 심사기간 연장요구의 건 등).
    conference_id   INTEGER NOT NULL REFERENCES meetings(conference_id) ON DELETE CASCADE,
    agenda_seq      INTEGER NOT NULL,        -- 회의록에 적힌 안건 번호 (1, 2, … 122)
    title           TEXT NOT NULL,
    bill_no         TEXT,
                    -- 의안이 아닌 안건은 NULL.
                    -- ⚠️ bills FK 를 걸지 않는다: 22대 밖 의안이나 아직 안 받은 의안을 가리킬 수
                    --    있다. 실측으로는 37,364건 전부 해소됐지만, 회의록이 의안보다 먼저 들어오는
                    --    수집 순서가 가능하므로 전방 참조를 막지 않는다.
    PRIMARY KEY (conference_id, agenda_seq)
);

CREATE INDEX IF NOT EXISTS idx_agenda_bill ON meeting_agenda(bill_no);

-- ═══════════════════════════════════════════════════════════
-- 수집 잔여물 — corpus 밖이다. 조회 질의는 이 테이블을 모른다
-- ═══════════════════════════════════════════════════════════

CREATE TABLE IF NOT EXISTS collect_failures (
    -- 받으려다 실패한 것들. 성공하면 행이 삭제되므로 **비어 있는 것이 정상 상태다.**
    target_kind     TEXT NOT NULL
                    CHECK (target_kind IN ('bill_detail','bill_vote','bill_proposer','bill_alt',
                                           'meeting_body','member')),
    target_key      TEXT NOT NULL,           -- bill_no · conference_id · open_na_id
    kind            TEXT NOT NULL CHECK (kind IN ('retriable','gone')),
                    -- 'gone' = 원천에 없다(404 등). 재시도하지 않는 것이 정상이다.
    detail          TEXT,                    -- 'network:ReadTimeout' · 'parse:no-caption'
                    -- 진단의 흔적이다. network: 가 몰리면 전송 문제, parse: 가 몰리면 DOM 변경.
                    -- ⚠️ **예외 타입만 적지 마라.** 한 함수가 서로 다른 이유로 같은 타입을 던지면
                    --    원장이 둘을 구분하지 못해 진단이 한 걸음 늦는다. 메시지까지 넣어라.
    attempts        INTEGER NOT NULL DEFAULT 0,
    last_attempt_at TEXT,
    PRIMARY KEY (target_kind, target_key)
    -- ⚠️ **"이미 받았는가"만으로 할 일을 정하면 안 된다.** 한 번 성공한 뒤의 재수집이 실패하면
    --    데이터는 남아 있으므로 "받았다"로 보이는데 원장에는 실패가 남는다. 그러면 그 항목은
    --    **영영 재시도되지 않고** 게이트가 영구히 빨갛다. 원장이 곧 재시도 큐다.
);
```

## 관계도

```
committees ─┬─< bills.committee            "이 위원회가 심사하는 법안"
            ├─< bill_stages.committee_name "이 위원회를 거친 법안"
            ├─< meetings.committee_name    "이 위원회의 회의"
            ├─< member_committees          "이 위원회의 의원"
            └─< committees.parent_committee (자기참조: 소위 → 상위)

members ────┬─< bill_proposers    "이 의원이 발의한 법안"
            ├─< bill_votes        "이 의원이 던진 표"
            ├─< member_committees "이 의원의 소속 위원회"
            └─< meeting_utterances.open_na_id  "이 의원의 발언"

bills ──────┬─< bill_stages            "어디에 언제 들어가 언제 나왔나"
            ├─< bill_proposers         "누가 냈나"
            ├─< bill_vote_summary(1:1) "표결 숫자"
            ├─< bill_votes             "누가 어떻게 던졌나"
            ├─< bill_promulgated_laws  "무슨 법이 됐나"
            ├─< bill_meetings ──> meetings ──< meeting_utterances   "무슨 말이 오갔나"
            ├─< meeting_agenda.bill_no (FK 없음)
            └── bills.alt_bill_no (자기참조, FK 없음: 전방 참조)

meetings ───┬─< meeting_utterances  "발언 + 화자"
            └─< meeting_agenda      "그날 다룬 안건"
```

## 대표 질의

새 스키마가 실제로 자명한지 보이는 예들이다. **구현 후 이 질의들이 그대로 도는지 확인한다.**

```sql
-- 이 법안 지금 어디까지 왔나
SELECT title, current_stage, outcome, committee FROM bills WHERE bill_no = ?;

-- 법사위에 묶여 있는 법안
SELECT bill_no, title FROM bills WHERE outcome='계류' AND current_stage='체계자구 심사';

-- 1년 넘게 소위에 멈춰 있는 법안
SELECT b.bill_no, b.title, s.committee_name, s.date_referred
FROM bills b JOIN bill_stages s ON s.bill_no = b.bill_no
WHERE b.outcome='계류' AND s.stage='소위'
  AND s.date_processed IS NULL AND s.date_referred < date('now','-1 year');

-- 대통령이 거부권을 쓴 법안과 그 결말
SELECT bill_no, title, date_reexam_requested, outcome FROM bills
WHERE date_reexam_requested IS NOT NULL;

-- 이 대안이 흡수한 원안들과 그 대표발의자
SELECT o.bill_no, o.title, m.name, m.party
FROM bills o JOIN bill_proposers p ON p.bill_no = o.bill_no AND p.role='대표발의'
JOIN members m ON m.open_na_id = p.open_na_id
WHERE o.alt_bill_no = ?;

-- 이 법안이 다뤄진 회의에서 오간 말
SELECT mt.date_meeting, mt.committee_name, u.speaker_name, u.speaker_position, u.text
FROM bill_meetings bm
JOIN meetings mt ON mt.conference_id = bm.conference_id
JOIN meeting_utterances u ON u.conference_id = mt.conference_id
```
